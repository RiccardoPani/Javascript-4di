let n1 = Number(prompt("Inserisci il primo numero:")); let operazione = prompt("Inserisci +, -, * oppure /"); let n2 = Number(prompt("Inserisci il secondo numero:")); let risultato; if (operazione === "+") { risultato = n1 + n2; } else if (operazione === "-") { risultato = n1 - n2; } else if (operazione === "*") { risultato = n1 * n2; } else if (operazione === "/") { risultato = n1 / n2; } else { risultato = "Operazione non valida"; } alert("Il risultato è: " + risultato);
